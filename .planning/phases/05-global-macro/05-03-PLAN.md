    ```python
    def _ticker_result(ticker: str, change_pct: float) -> dict:
        """Return a successful ticker result dict."""
        return {"change_pct": change_pct}
    
    def _ticker_error(ticker: str, msg: str = "mock error") -> dict:
        """Return a failed ticker result dict."""
        return {"error": msg}
    
    def _all_success(changes: dict[str, float]) -> dict[str, dict]:
        """Build a full fetch result dict with all tickers succeeding."""
        return {t: _ticker_result(t, changes.get(t, 0.0)) for t in GLOBAL_TICKERS}
    
    def _some_failed(successful: dict[str, float]) -> dict[str, dict]:
        """Build fetch result with only specified successful tickers."""
        result = {}
        for t, w in GLOBAL_TICKERS.items():
            if t in successful:
                result[t] = _ticker_result(t, successful[t])
            else:
                result[t] = _ticker_error(t)
        return result
    ```
