    ```python
    """OICollector — polls Nifty/BankNifty futures OI every 300s via AngelOneClient.

    Tracks OI change percentage against previous poll value stored in Redis.
    OI buildup suggests institutional conviction; OI unwind suggests weakening.
    """

    from __future__ import annotations

    import asyncio
    from datetime import datetime, timezone
    from typing import Any

    from variance.base_collector import BaseVarianceCollector
    from variance.collectors._angel import _get_angel_client
    from variance.schemas import ParseResult

    # Redis key pattern for OI baselines (per D-08)
    OI_BASELINE_KEY_PREFIX = "mve:oi_baseline"

    # Tracked symbols (Nifty and BankNifty futures)
    TRACKED_SYMBOLS = ["NIFTY", "BANKNIFTY"]


    class OICollector(BaseVarianceCollector):
        """Poll Nifty/BankNifty futures OI every 300s via AngelOneClient.

        Per D-07: fetch() calls get_futures_oi() for each symbol.
        Per D-08: OI change computed against previous poll stored in Redis.
        Per D-09: OI scoring threshold-based with linear interpolation.
        """

        def __init__(self, redis_cache=None) -> None:
            super().__init__(name="oi", poll_interval=300)
            self._redis = redis_cache

        async def fetch(self) -> Any:
            """Fetch futures OI for all tracked symbols via AngelOneClient.

            Per D-07: calls await asyncio.to_thread(self._angel.get_futures_oi, symbol).
            Per D-03: AngelOneClient calls wrapped in asyncio.to_thread.
            """
            angel = _get_angel_client()
            results: dict[str, dict[str, Any]] = {}
            for symbol in TRACKED_SYMBOLS:
                try:
                    data = await asyncio.to_thread(angel.get_futures_oi, symbol)
                    results[symbol] = data if isinstance(data, dict) else {}
                except Exception:
                    results[symbol] = {}
            return results

        def parse(self, raw: Any) -> ParseResult:
            """Parse futures OI data for all tracked symbols.

            Computes total OI across all symbols and tracks individual
            symbol OI values for detailed analysis.

            Expected raw shape:
            {
                "NIFTY": {"open_interest": 12345678, "ltp": 23456.0, ...},
                "BANKNIFTY": {"open_interest": 9876543, "ltp": 49876.0, ...},
            }
            """
            if not isinstance(raw, dict):
                raise ValueError(f"Unexpected OI data type: {type(raw)}")

            symbols_detail: dict[str, Any] = {}
            total_oi: float = 0.0
            symbols_with_data: int = 0

            for symbol in TRACKED_SYMBOLS:
                data = raw.get(symbol, {})
                if isinstance(data, dict) and data.get("open_interest") is not None:
                    oi = self._to_float(data["open_interest"], 0.0)
                    ltp = self._to_float(data.get("ltp"), 0.0)
                    symbols_detail[symbol] = {
                        "open_interest": oi,
                        "ltp": ltp,
                        "has_data": True,
                    }
                    total_oi += oi
                    symbols_with_data += 1
                else:
                    symbols_detail[symbol] = {
                        "open_interest": 0.0,
                        "ltp": 0.0,
                        "has_data": False,
                    }

            if symbols_with_data == 0:
                raise ValueError("No OI data received for any tracked symbol")

            detail: dict[str, Any] = {
                "symbols": symbols_detail,
                "total_oi": total_oi,
                "symbols_with_data": symbols_with_data,
                "tracked_count": len(TRACKED_SYMBOLS),
            }

            return ParseResult(
                raw_value=total_oi,
                normalized=0.0,  # Placeholder — score() sets this
                direction=0,     # Direction determined by OI change vs baseline
                magnitude=0.0,   # Magnitude scaled by OI change %
                detail=detail,
                source="angel",
                as_of=datetime.now(timezone.utc).isoformat(),
            )

        def score(self, parsed: ParseResult) -> float:
            """Score OI change percentage.

            Per D-09:
            - OI change % computed against previous poll (stored in Redis)
            - >3% buildup → +0.3
            - < -3% unwind → -0.3
            - Linear interpolation between -3% and +3% (0% change → 0.0)

            The score is derived from the OI change percentage stored in the
            detail dict. The change % must be computed externally (via Redis)
            and placed in detail["oi_change_pct"] before score() is called.

            This design keeps score() stateless and testable — the Redis
            comparison happens in the caller (poll() override or external
            orchestration).
            """
            detail = parsed.get("detail", {})
            oi_change_pct = detail.get("oi_change_pct", 0.0)

            if oi_change_pct <= -3.0:
                return -0.3
            elif oi_change_pct >= 3.0:
                return 0.3
            else:
                # Linear interpolation: -3%→-0.3, 0%→0.0, 3%→0.3
                return oi_change_pct / 10.0

        async def poll_with_baseline(
            self,
            redis_cache,
        ) -> ParseResult:
            """Full poll cycle with Redis baseline comparison.

            This is the integration-aware version of poll() that:
            1. Fetches current OI data
            2. Loads previous OI baseline from Redis
            3. Computes OI change percentage
            4. Stores new OI baseline in Redis
            5. Calls score() with the computed change

            Per D-08: OI baseline stored under mve:oi_baseline:{symbol}
            with 1-hour TTL. First poll returns score 0.0.
            """
            raw = await self.fetch()
            parsed = self.parse(raw)

            total_oi = parsed["detail"]["total_oi"]
            oi_change_pct = 0.0

            if redis_cache:
                baseline_key = f"{OI_BASELINE_KEY_PREFIX}:total"
                previous = await redis_cache.get_mve(baseline_key)

                if previous is not None and "total_oi" in previous:
                    prev_oi = self._to_float(previous["total_oi"], 0.0)
                    if prev_oi > 0:
                        oi_change_pct = round((total_oi - prev_oi) / prev_oi * 100, 2)

                    # Store per-symbol baselines
                    for symbol in TRACKED_SYMBOLS:
                        sym_key = f"{OI_BASELINE_KEY_PREFIX}:{symbol}"
                        sym_data = parsed["detail"]["symbols"].get(symbol, {})
                        if sym_data.get("has_data"):
                            prev_sym = await redis_cache.get_mve(sym_key)
                            if prev_sym is not None and "open_interest" in prev_sym:
                                prev_oi = self._to_float(prev_sym["open_interest"], 0.0)
                                if prev_oi > 0:
                                    sym_change = round(
                                        (sym_data["open_interest"] - prev_oi) / prev_oi * 100, 2
                                    )
                                    parsed["detail"]["symbols"][symbol]["oi_change_pct"] = sym_change

                # Store new baseline (TTL=3600 = 1 hour per D-08)
                await redis_cache.set_mve(
                    baseline_key,
                    {"total_oi": total_oi, "as_of": parsed["as_of"]},
                    ttl=3600,
                )
                for symbol in TRACKED_SYMBOLS:
                    sym_data = parsed["detail"]["symbols"].get(symbol, {})
                    if sym_data.get("has_data"):
                        await redis_cache.set_mve(
                            f"{OI_BASELINE_KEY_PREFIX}:{symbol}",
                            {"open_interest": sym_data["open_interest"], "as_of": parsed["as_of"]},
                            ttl=3600,
                        )

            else:
                # No Redis connection — linear mode (no baseline comparison)
                oi_change_pct = 0.0

            # Update parsed result with OI change data
            parsed["detail"]["oi_change_pct"] = oi_change_pct
            score_val = self.score(parsed)
            parsed["normalized"] = score_val
            parsed["direction"] = 1 if score_val > 0 else (-1 if score_val < 0 else 0)
            parsed["magnitude"] = abs(score_val)

            # Update base class state
            self._last_successful_result = parsed  # type: ignore
            self._last_poll_time = datetime.now(timezone.utc)
            self._consecutive_errors = 0

            return parsed

        @staticmethod
        def _to_float(value: Any, default: float | None = None) -> float | None:
            """Safely convert a value to float, returning default on failure."""
            if value is None:
                return default
            try:
                return float(value)
            except (TypeError, ValueError):
                return default
    ```
