    """Shared AngelOneClient singleton for MVE collectors.

    Lazily initialized on first use — import does NOT create a session or authenticate.
    Both FIIDIICollector and OICollector share the same instance via _get_angel_client().

    Per D-04: matches _nse.py singleton pattern.
    """

    from __future__ import annotations

    import asyncio
    import logging
    from typing import Any

    from data.collector.angel_client import AngelOneClient

    _logger = logging.getLogger(__name__)

    _angel_client: AngelOneClient | None = None
    _angel_config: dict[str, Any] | None = None


    def _set_angel_config(config: dict[str, Any]) -> None:
        """Set Angel One config before first client initialization.

        Must be called before _get_angel_client() if not using env vars.
        Config dict should contain keys: api_key, client_id, password, totp_secret.
        """
        global _angel_config
        _angel_config = config


    def _get_angel_client() -> AngelOneClient:
        """Return the shared AngelOneClient instance, lazy-initializing on first call."""
        global _angel_client
        if _angel_client is None:
            _logger.info("Initializing shared AngelOneClient instance")
            cfg = _angel_config or {}
            _angel_client = AngelOneClient(cfg)
        return _angel_client
    ```

    """FIIDIICollector — polls FII/DII net flows every 30min via NseIndiaApi."""

    from __future__ import annotations

    from datetime import datetime, timezone
    from typing import Any

    from variance.base_collector import BaseVarianceCollector
    from variance.collectors._nse import _fetch_fii_dii_data
    from variance.schemas import ParseResult


    class FIIDIICollector(BaseVarianceCollector):
        """Poll FII/DII net flows every 1800s (30min).

        Per D-05: fetch() calls _fetch_fii_dii_data() via NseIndiaApi.
        Per D-06: FII weighted 0.7, DII weighted 0.3, normalized to ±4000Cr.
        """

        def __init__(self) -> None:
            super().__init__(name="fii_dii", poll_interval=1800)

        async def fetch(self) -> Any:
            """Fetch FII/DII net flows data from NseIndiaApi."""
            return await _fetch_fii_dii_data()

        def parse(self, raw: Any) -> ParseResult:
            """Extract FII and DII net flow values from NseIndiaApi response.

            Expected data shape from NseIndiaApi varies by version.
            Common shapes:
              {"fii_net": 1234.5, "dii_net": -567.8}
              {"fii": {"buy": ..., "sell": ..., "net": 1234.5}, "dii": {...}}
              {"data": [{"category": "FII", "net": 1234.5}, {"category": "DII", "net": -567.8}]}

            Per 03-CONTEXT.md Claude's Discretion: handle the actual response shape
            during implementation. The parse() method must handle at least one common shape.
            """
            if not isinstance(raw, dict):
                raise ValueError(f"Unexpected FII/DII data type: {type(raw)}")

            fii_net: float | None = None
            dii_net: float | None = None

            # Shape 1: {"fii_net": ..., "dii_net": ...}
            if "fii_net" in raw or "FII_NET" in raw:
                fii_net = self._to_float(raw.get("fii_net") or raw.get("FII_NET"))
                dii_net = self._to_float(raw.get("dii_net") or raw.get("DII_NET"))

            # Shape 2: {"fii": {"net": ...}, "dii": {"net": ...}}
            elif "fii" in raw or "FII" in raw:
                fii_data = raw.get("fii") or raw.get("FII") or {}
                dii_data = raw.get("dii") or raw.get("DII") or {}
                fii_net = self._to_float(fii_data.get("net") or fii_data.get("NET"))
                dii_net = self._to_float(dii_data.get("net") or dii_data.get("NET"))

            # Shape 3: NseIndiaApi may use different keys — extract what we can
            else:
                # Try numeric values from any discovered keys
                for key, val in raw.items():
                    if isinstance(val, (int, float)):
                        if "fii" in key.lower():
                            fii_net = float(val)
                        elif "dii" in key.lower():
                            dii_net = float(val)

            if fii_net is None:
                raise ValueError("FII net flow not found in NseIndiaApi response")

            combined_net = (fii_net * 0.7) + ((dii_net or 0.0) * 0.3)

            return ParseResult(
                raw_value=combined_net,
                normalized=0.0,  # Placeholder — score() sets this
                direction=1 if combined_net > 0 else (-1 if combined_net < 0 else 0),
                magnitude=min(1.0, abs(combined_net) / 4000.0),
                detail={
                    "fii_net": fii_net,
                    "dii_net": dii_net or 0.0,
                    "combined_net": combined_net,
                    "fii_weight": 0.7,
                    "dii_weight": 0.3,
                },
                source="nse",
                as_of=datetime.now(timezone.utc).isoformat(),
            )

        def score(self, parsed: ParseResult) -> float:
            """Score FII/DII net flows normalized to [-1.0, 1.0].

            Per D-06:
            - Combined net = FII(0.7) + DII(0.3)
            - ±4000Cr → ±1.0 (linear normalization)
            - Clamp beyond ±4000Cr to ±1.0
            """
            combined = parsed["raw_value"]
            normalized = combined / 4000.0
            return max(-1.0, min(1.0, normalized))

        @staticmethod
        def _to_float(value: Any, default: float | None = None) -> float | None:
            """Safely convert a value to float, returning default on failure."""
            if value is None:
                return default
            try:
                return float(value)
            except (TypeError, ValueError):
                return default
    ```
