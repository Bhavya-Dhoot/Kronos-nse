    """InstitutionalDimensionAggregator — combines FII/DII and OI dimension scores."""

    from __future__ import annotations

    from datetime import datetime, timezone
    from typing import Any

    from variance.schemas import DimensionScore


    class InstitutionalDimensionAggregator:
        """Combine FII/DII (0.7) and OI (0.3) into a single institutional dimension score.

        Per D-10: standalone dataclass-like class (not a collector).
        Per D-11: compute() returns a DimensionScore TypedDict ready for
        MarketVarianceScore.build().

        The aggregator does NOT poll data — it computes from scores produced
        by FIIDIICollector and OICollector. This separation of concerns means
        the aggregator can be tested without any network calls.
        """

        def __init__(self) -> None:
            self.fii_dii_score: float | None = None
            self.oi_score: float | None = None
            self.fii_dii_stale: bool = False
            self.oi_stale: bool = False

        def compute(
            self,
            fii_dii_score: float | None = None,
            oi_score: float | None = None,
            fii_dii_stale: bool = False,
            oi_stale: bool = False,
        ) -> DimensionScore:
            """Compute the combined institutional dimension score.

            Args:
                fii_dii_score: Score from FIIDIICollector [-1.0, 1.0], or None if unavailable.
                oi_score: Score from OICollector [-1.0, 1.0], or None if unavailable.
                fii_dii_stale: True if FII/DII data is stale (not from fresh poll).
                oi_stale: True if OI data is stale.

            Returns:
                DimensionScore with combined score, weights reflected in score calculation.

            Formula (per D-10):
                combined = fii_dii_score * 0.7 + oi_score * 0.3
                If one dimension is None, the other gets full weight.
                Clamped to [-1.0, 1.0].
            """
            self.fii_dii_score = fii_dii_score
            self.oi_score = oi_score
            self.fii_dii_stale = fii_dii_stale
            self.oi_stale = oi_stale

            if fii_dii_score is not None and oi_score is not None:
                combined = fii_dii_score * 0.7 + oi_score * 0.3
                active_weight = 1.0
            elif fii_dii_score is not None:
                combined = fii_dii_score
                active_weight = 0.7
            elif oi_score is not None:
                combined = oi_score
                active_weight = 0.3
            else:
                combined = 0.0
                active_weight = 0.0

            combined = max(-1.0, min(1.0, combined))

            detail: dict[str, Any] = {
                "fii_dii_score": fii_dii_score,
                "oi_score": oi_score,
                "fii_dii_stale": fii_dii_stale,
                "oi_stale": oi_stale,
                "active_weight": active_weight,
                "fii_dii_weight": 0.7,
                "oi_weight": 0.3,
            }

            return DimensionScore(
                name="institutional",
                score=round(combined, 4),
                weight=0.25,  # Per config/base.yaml §variance.weights.institutional
                is_stale=(fii_dii_score is None and oi_score is None),
                detail=detail,
                collected_at=datetime.now(timezone.utc).isoformat(),
            )
    ```

    Key design points:
    - Standalone class — NOT a collector (no fetch/parse/score methods, no BaseVarianceCollector base class)
    - `compute()` takes pre-computed scores and returns a DimensionScore (D-11)
    - Formula: FII/DII 0.7 + OI 0.3 (D-10), with graceful handling of missing values
    - When one dimension is None (not yet collected), the other gets full weight
    - `weight=0.25` matches `config/base.yaml §variance.weights.institutional`
    - Only marked `is_stale=True` when BOTH scores are None — partial data is still valid
    - The aggregator is consumed by Phase 6's MarketVarianceEngine._recompute_mvs()

    """Unit tests for InstitutionalDimensionAggregator.

    The aggregator is pure computation — no network calls, no mocking needed.
    """

    from __future__ import annotations

    import pytest

    from variance.aggregators.institutional import InstitutionalDimensionAggregator


    @pytest.fixture
    def aggregator() -> InstitutionalDimensionAggregator:
        return InstitutionalDimensionAggregator()


    class TestCompute:

        def test_combines_both_scores(self, aggregator):
            """FII/DII (0.7) * 0.5 + OI (0.3) * 0.3 = 0.35 + 0.09 = 0.44."""
            result = aggregator.compute(fii_dii_score=0.5, oi_score=0.3)
            assert result["score"] == pytest.approx(0.44, abs=0.001)
            assert result["name"] == "institutional"
            assert result["weight"] == 0.25

        def test_combines_opposite_signs(self, aggregator):
            """FII/DII positive, OI negative → mixed signal.
            (0.7)*0.7 + (0.3)*(-0.5) = 0.49 - 0.15 = 0.34."""
            result = aggregator.compute(fii_dii_score=0.7, oi_score=-0.5)
            assert result["score"] == pytest.approx(0.34, abs=0.001)

        def test_fii_dii_only(self, aggregator):
            """OI=None → FII/DII gets full weight."""
            result = aggregator.compute(fii_dii_score=0.5)
            assert result["score"] == pytest.approx(0.5, abs=0.001)
            assert result["detail"]["oi_score"] is None

        def test_oi_only(self, aggregator):
            """FII/DII=None → OI gets full weight."""
            result = aggregator.compute(oi_score=0.3)
            assert result["score"] == pytest.approx(0.3, abs=0.001)
            assert result["detail"]["fii_dii_score"] is None

        def test_both_none_returns_zero(self, aggregator):
            """No data → score = 0.0, is_stale = True."""
            result = aggregator.compute()
            assert result["score"] == 0.0
            assert result["is_stale"] is True  # Both None → stale

        def test_partial_data_not_stale(self, aggregator):
            """One dimension available → not stale."""
            result = aggregator.compute(fii_dii_score=0.5)
            assert result["is_stale"] is False

        def test_clamping_above_1(self, aggregator):
            """Combined > 1.0 → clamped to 1.0."""
            result = aggregator.compute(fii_dii_score=1.0, oi_score=1.0)
            # 1.0*0.7 + 1.0*0.3 = 1.0 → exactly at boundary
            assert result["score"] == pytest.approx(1.0, abs=0.001)

        def test_clamping_below_neg1(self, aggregator):
            """Combined < -1.0 → clamped to -1.0."""
            result = aggregator.compute(fii_dii_score=-1.0, oi_score=-1.0)
            # -1.0*0.7 + (-1.0)*0.3 = -1.0 → exactly at boundary
            assert result["score"] == pytest.approx(-1.0, abs=0.001)

        def test_detail_contains_all_keys(self, aggregator):
            """Detail dict has all expected metadata."""
            result = aggregator.compute(fii_dii_score=0.5, oi_score=0.3)
            detail = result["detail"]
            assert detail["fii_dii_score"] == 0.5
            assert detail["oi_score"] == 0.3
            assert detail["fii_dii_weight"] == 0.7
            assert detail["oi_weight"] == 0.3
            assert detail["active_weight"] == 1.0
            assert "collected_at" in result

        def test_stale_flags_passed_through(self, aggregator):
            """Stale flags propagate to detail."""
            result = aggregator.compute(
                fii_dii_score=0.5, oi_score=0.3,
                fii_dii_stale=True, oi_stale=False,
            )
            assert result["detail"]["fii_dii_stale"] is True
            assert result["detail"]["oi_stale"] is False
    ```

    - No mocking needed — pure computation tests
    - Verify both-scores, single-score, no-score combinations
    - Verify clamping at ±1.0
    - Verify detail dict contains all metadata
    - Verify stale flag propagation
