```python
class TestVarianceWebSocket:
    """WS /ws/variance"""

    def test_ws_connects_and_receives_ping(self, client: TestClient):
        with client.websocket_connect("/ws/variance") as ws:
            msg = ws.receive_json()
            assert msg["type"] == "ping"

    def test_ws_receives_mvs_update_on_redis_message(self, client: TestClient):
        import json

        # Set up pubsub to simulate a Redis mvs message
        pubsub = AsyncMock()
        entry = json.dumps(_sample_mvs_dict())
        pubsub.subscribe = AsyncMock()
        pubsub.get_message = AsyncMock(side_effect=[
            None,  # First call returns None (subscribe message filtered)
            {      # Second call returns actual MVS message
                "type": "message",
                "data": entry,
                "channel": "mve:mvs:updates",
            },
        ])
        pubsub.unsubscribe = AsyncMock()
        pubsub.aclose = AsyncMock()
        client.app.state.mve_redis.pubsub.return_value = pubsub

        with client.websocket_connect("/ws/variance") as ws:
            # First message is the ping
            ping = ws.receive_json()
            assert ping["type"] == "ping"

            # Second message should be the transformed MVS update
            update = ws.receive_json(timeout=5)
            assert update["type"] == "mvs_update"
            assert "payload" in update
            assert update["payload"]["composite"] == 0.35
            assert update["payload"]["market_state"] == "bull_run"

    def test_ws_errors_when_mve_not_available(self, client: TestClient):
        client.app.state.mve_redis = None
        with client.websocket_connect("/ws/variance") as ws:
            msg = ws.receive_json()
            assert msg["type"] == "error"
            assert "MVE not available" in msg["detail"]
```

**Key test behaviors:**
1. Ping on connect — confirms WS accepts connections
2. MVS update forwarding — simulates Redis pubsub message, confirms typed `{"type": "mvs_update", "payload": {...}}` format
3. Error handling — confirms graceful error when MVE not available (D-11/D-12)

Use `timeout=5` on receive_json to allow for async processing. The `side_effect` on `get_message` returns None first (subscribe message filtered) then the actual MVS payload.
